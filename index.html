// 1. Año automático en el pie de página document.getElementById("anio").textContent = new Date().getFullYear(); // 2. Menú para celular const menuBtn = document.getElementById("menuBtn"); const menu = document.getElementById("menu"); menuBtn.addEventListener("click", () => { menu.classList.toggle("abierto"); }); menu.querySelectorAll("a").forEach((enlace) => { enlace.addEventListener("click", () => menu.classList.remove("abierto")); }); // 3. Si la foto no carga, mostrar las iniciales const foto = document.getElementById("foto"); const iniciales = document.getElementById("iniciales"); foto.addEventListener("error", () => { foto.style.display = "none"; iniciales.style.display = "flex"; }); // 4. Efecto de escritura en el título const texto = "Técnica en Informática y Desarrollo Web"; const escribiendo = document.getElementById("escribiendo"); let i = 0; function escribir() { if (i <= texto.length) { escribiendo.textContent = texto.slice(0, i); i++; setTimeout(escribir, 60); } } escribir(); // 5. Las secciones aparecen suavemente al bajar const secciones = document.querySelectorAll(".oculto"); const observador = new IntersectionObserver( (entradas) => { entradas.forEach((entrada) => { if (entrada.isIntersecting) { entrada.target.classList.add("visible"); } }); }, { threshold: 0.15 } ); secciones.forEach((s) => observador.observe(s)); // 6. Resaltar en el menú la sección donde estás const enlaces = document.querySelectorAll("nav a"); window.addEventListener("scroll", () => { let actual = ""; document.querySelectorAll("main section").forEach((sec) => { if (window.scrollY >= sec.offsetTop - 120) { actual = sec.id; } }); enlaces.forEach((a) => { a.classList.toggle("activo", a.getAttribute("href") === "#" + actual); }); });
